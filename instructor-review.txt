function getInstructorApiToken() {
    return (
        window.SIKNAYAN_API_TOKEN ||

        localStorage.getItem("token") ||
        localStorage.getItem("access_token") ||

async function instructorApiRequest(
    path,
    options = {}
) {
    const token =
        getInstructorApiToken();

    if (!token) {
        throw new Error(
            "No API token found. Please save a fresh Laravel token as siknayan_api_token."
        );
    }

    const response =
        await fetch(
            `${INSTRUCTOR_API_BASE_URL}${path}`,
            {
                ...options,

                headers: {
                    Accept:
                        "application/json",

                    "Content-Type":
                        "application/json",

                    ...(options.headers || {}),

                    Authorization:
                        `Bearer ${token}`
                }
            }
        );

    let payload = null;

    try {
        payload =
            await response.json();
    } catch {
        payload = null;
    }

    if (!response.ok) {
        throw new Error(
            payload?.message ||
            `API request failed (${response.status}).`
        );
    }

    return payload;
}

    function escapeHtml(
        value
    ) {

        return String(
            value ?? ""
        )
            .replaceAll(
                "&",
                "&amp;"
            )
            .replaceAll(
                "<",
                "&lt;"
            )
            .replaceAll(
                ">",
                "&gt;"
            )
            .replaceAll(
                '"',
                "&quot;"
            )
            .replaceAll(
                "'",
                "&#039;"
            );

    }

    function createInstructorId() {

        const instructors =
            loadInstructors();

        let highestNumber = 0;

        instructors.forEach(
            instructor => {

                const match =
                    String(
                        instructor.id || ""
                    ).match(
                        /^INS-(\d+)$/
                    );

                if (!match) {
                    return;
                }

                highestNumber =
                    Math.max(
                        highestNumber,
                        Number(match[1])
                    );

            }
        );

        return `INS-${String(
            highestNumber + 1
        ).padStart(3, "0")}`;

    }


    async function syncInstructorsFromApi() {

        const payload =
            await instructorApiRequest(
                "/admin/instructors"
            );

        const apiInstructors =
            Array.isArray(payload?.instructors)
                ? payload.instructors
                : [];

        const instructors =
            apiInstructors.map(
                instructor => ({

                    id:
                        instructor.school_id ||
                        instructor.id,

                    backendId:
                        instructor.id,

                    name:
                        instructor.name ||
                        "Unnamed Instructor",

                    email:
                        instructor.email ||
                        "",

                    schoolId:
                        instructor.school_id ||
                        "",

                    program:
                        instructor.program ||
                        "",

                    contact:
                        instructor.contact ||
                        "",

                    status:
                        "active",

                    passwordSetupComplete:
                        Boolean(
                            instructor.password_setup_complete
                        )

                })
            );

        saveInstructors(
            instructors
        );


        /*
         * Build the account view from the
         * Laravel instructor records.
         *
         * Do NOT store passwords.
         */
        const accounts =
            apiInstructors.map(
                instructor => ({

                    id:
                        `instructor-account-${instructor.id}`,

                    instructorId:
                        instructor.school_id ||
                        instructor.id,

                    backendId:
                        instructor.id,

                    email:
                        instructor.email ||
                        "",

                    lastLogin:
                        "Never",

                    status:
                        "active",

                    backendManaged:
                        true

                })
            );

        saveAccounts(
            accounts
        );

        return instructors;

    }
    


    function generateTemporaryPassword() {

        const characters =
            "ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz23456789";

        let password =
            "";


        for (
            let index = 0;
            index < 10;
            index++
        ) {

            password +=
                characters[
                    Math.floor(
                        Math.random() *
                        characters.length
                    )
                ];

        }


        return password;

    }

    function loadInstructors() {

        try {

            const stored =
                localStorage.getItem(
                    INSTRUCTOR_STORAGE_KEY
                );


            if (stored) {

                const parsed =
                    JSON.parse(
                        stored
                    );


                if (
                    Array.isArray(
                        parsed
                    )
                ) {

                    return parsed;

                }

            }

        } catch (error) {

            console.error(
                "Instructor storage error:",
                error
            );

        }


        /*
         * Only use class data as a fallback
         * when instructor records do not exist.
         */

        const classes =
            loadClasses();

        const instructors =
            [];


        classes.forEach(
            classItem => {

                if (
                    !classItem.instructorId
                ) {

                    return;

                }


                const exists =
                    instructors.some(
                        instructor =>
                            String(
                                instructor.id
                            ) ===
                            String(
                                classItem.instructorId
                            )
                    );


                if (exists) {

                    return;

                }


                instructors.push({

                    id:
                        classItem.instructorId,

                    name:
                        classItem.instructorName ||
                        "Unnamed Instructor",

                    status:
                        "active"

                });

            }
        );


        if (
            instructors.length
        ) {

            saveInstructors(
                instructors
            );

        }


        return instructors;

    }

       async function saveAccount(
        event
    ) {

        event.preventDefault();

        const instructorId =
            instructorIdInput.value
                .trim();

        const instructorName =
            instructorNameInput.value
                .trim();

        const email =
            emailInput.value
                .trim()
                .toLowerCase();

        const password =
            passwordInput.value
                .trim();

        const status =
            accountStatusInput.value ||
            "active";

        const isEditing =
            Boolean(editingId);


        /*
         * Instructor ID, name, and email are always required.
         * Password is required only when creating a new
         * instructor account. Laravel handles hashing.
         */
        if (
            !instructorId ||
            !instructorName ||
            !email ||
            (!isEditing && !password)
        ) {

            alert(
                "Please complete all required fields."
            );

            return;

        }


        if (
            !isEditing &&
            password.length < 8
        ) {

            alert(
                "Password must be at least 8 characters."
            );

            return;

        }


        /*
         * Prevent duplicate email addresses in the
         * currently synchronized instructor records.
         */
        const accounts =
            getAccounts();

        const duplicateEmail =
            accounts.some(
                account =>
                    String(
                        account.email ||
                        ""
                    ).toLowerCase() ===
                    email &&
                    account.id !==
                    editingId
            );


        if (
            duplicateEmail
        ) {

            alert(
                "That email address is already being used by another instructor account."
            );

            return;

        }


        try {

            /* =================================================
               EDIT EXISTING INSTRUCTOR
            ================================================== */

            if (
                isEditing
            ) {

                const account =
                    accounts.find(
                        item =>
                            item.id ===
                            editingId
                    );


                const backendId =
                    account?.backendId;


                if (
                    !backendId
                ) {

                    throw new Error(
                        "This instructor is not linked to a Laravel account. Refresh the page and try again."
                    );

                }


                const requestData = {
                    name:
                        instructorName,

                    email:
                        email
                };


                /*
                 * Only send a password when the administrator
                 * actually entered one during editing.
                 */
                if (
                    password
                ) {

                    if (
                        password.length < 8
                    ) {

                        alert(
                            "Password must be at least 8 characters."
                        );

                        return;

                    }

                    requestData.password =
                        password;

                }


                const payload =
                    await instructorApiRequest(
                        `/admin/instructors/${backendId}`,
                        {
                            method:
                                "PUT",

                            body:
                                JSON.stringify(
                                    requestData
                                )
                        }
                    );


                /*
                 * Re-sync from Laravel so the UI reflects
                 * the actual database state.
                 */
                await syncInstructorsFromApi();

                closeAccountModal();

                renderTable();

                refreshPageIcons();

                alert(
                    payload?.message ||
                    "Instructor updated successfully."
                );

                return;

            }


            /* =================================================
               CREATE NEW INSTRUCTOR
            ================================================== */

            const payload =
                await instructorApiRequest(
                    "/admin/instructors",
                    {
                        method:
                            "POST",

                        body:
                            JSON.stringify({
                                name:
                                    instructorName,

                                email:
                                    email,

                                school_id:
                                    instructorId,

                                password:
                                    password
                            })
                    }
                );


            /*
             * Laravel is now the source of truth.
             * Do not store the submitted password.
             */
            await syncInstructorsFromApi();

            closeAccountModal();

            renderTable();

            refreshPageIcons();


            alert(
                payload?.message ||
                `Instructor account created for ${instructorName}.`
            );

        } catch (
            error
        ) {

            console.error(
                "Instructor save failed:",
                error
            );

            alert(
                error.message ||
                "Unable to save instructor."
            );

        }

    }



    /* =====================================================

    function renderTable() {

        if (!tableBody) {

            return;

        }


        const instructors =
            loadInstructors();


        const query =
            searchInput?.value
                .trim()
                .toLowerCase() ||
            "";


        const selectedStatus =
            statusFilter?.value ||
            "all";


        const rows =
            [];


        /*
         * The instructor is the row entity.
         * Account data is attached to the instructor
         * when access has been created.
         */

        instructors.forEach(
            instructor => {

                const account =
                    getAccountForInstructor(
                        instructor.id
                    );


                const assignedClasses =
                    getAssignedClasses(
                        instructor.id
                    );


                const accountStatus =
                    account
                        ? String(
                            account.status ||
                            "active"
                        ).toLowerCase()
                        : "not-set-up";


                const searchableText = [

                    instructor.id,

                    instructor.name,

                    account?.email,

                    ...assignedClasses.map(
                        getClassDisplayName
                    )

                ]
                    .join(" ")
                    .toLowerCase();


                if (
                    query &&
                    !searchableText.includes(
                        query
                    )
                ) {

                    return;

                }


                if (
                    selectedStatus !==
                    "all" &&
                    accountStatus !==
                    selectedStatus
                ) {

                    return;

                }


                rows.push({

                    instructor,

                    account,

                    assignedClasses,

                    accountStatus

                });

            }
        );


        tableBody.innerHTML =
            "";


        if (
            emptyState
        ) {

            emptyState.hidden =
                rows.length !==
                0;

        }



        /* =================================================
           RENDER EACH INSTRUCTOR
        ================================================== */

        rows.forEach(
            ({
                instructor,
                account,
                assignedClasses,
                accountStatus
            }) => {

                const row =
                    document.createElement(
                        "tr"
                    );


                row.className =
                    "learner-account-row";



                /* -----------------------------------------
                   INSTRUCTOR ID
                ----------------------------------------- */

                const instructorIdCell =
                    document.createElement(
                        "td"
                    );


               instructorIdCell.textContent =
                    instructor.schoolId ||
                    instructor.id;


                /* -----------------------------------------
                   INSTRUCTOR
                ----------------------------------------- */

                const instructorCell =
                    document.createElement(
                        "td"
                    );


                instructorCell.className =
                    "learner-account-name";


                instructorCell.innerHTML =
                    `
                    <strong>
                        ${escapeHtml(
                            instructor.name
                        )}
                    </strong>
                    `;



                /* -----------------------------------------
                   EMAIL
                ----------------------------------------- */

                const emailCell =
                    document.createElement(
                        "td"
                    );


                emailCell.className =
                    "learner-account-email";


                emailCell.textContent =
                    account?.email ||
                    "—";



                /* -----------------------------------------
                   ASSIGNED CLASSES
                ----------------------------------------- */

                const classesCell =

 async function confirmResetPassword() {

    if (!resetAccountId) {
        return;
    }

    const accounts =
        getAccounts();

    const account =
        accounts.find(
            item =>
                item.id ===
                resetAccountId
        );

    if (!account) {
        alert(
            "Instructor account not found. Please refresh the page and try again."
        );
        return;
    }

    const password =
        resetPasswordValue.value.trim();

    if (!password) {
        alert(
            "Please generate a temporary password first."
        );
        return;
    }

    if (password.length < 8) {
        alert(
            "The temporary password must be at least 8 characters."
        );
        return;
    }

    const backendId =
        account.backendId;

    if (!backendId) {
        alert(
            "This instructor is not linked to a Laravel account. Refresh the page and try again."
        );
        return;
    }

    confirmResetPasswordButton.disabled =
        true;

    confirmResetPasswordButton.innerHTML = `
        <i data-lucide="loader-circle"></i>
        <span>Resetting...</span>
    `;

    refreshPageIcons();

    try {

        const payload =
            await instructorApiRequest(
                `/admin/instructors/${backendId}`,
                {
                    method: "PUT",

                    body: JSON.stringify({
                        password: password
                    })
                }
            );

        console.log(
            "Instructor password reset response:",
            payload
        );

        closeResetPasswordModal();

        confirmResetPasswordButton.disabled =
            false;

        confirmResetPasswordButton.innerHTML = `
            <i data-lucide="key-round"></i>
            <span>Reset Password</span>
        `;

        renderTable();
        refreshPageIcons();

        alert(
            payload?.message ||
            `The temporary password has been reset successfully.\n\nNew password: ${password}`
        );

    } catch (error) {

        console.error(
            "Instructor password reset failed:",
            error
        );

        confirmResetPasswordButton.disabled =
            false;

        confirmResetPasswordButton.innerHTML = `
            <i data-lucide="key-round"></i>
            <span>Reset Password</span>
        `;

        refreshPageIcons();

        alert(
            error.message ||
            "Unable to reset the instructor password."
        );
    }
}

    const instructorId =

    const instructorName =
        instructor.name ||
        "The instructor";

    /*
     * Find the Laravel instructor ID.
     *
     * The frontend instructor.id may be
     * INS-001, INS-002, etc., while
     * backendId is the actual users.id.
     */
    const account =
        getAccountForInstructor(
            instructor.id
        );

    const backendId =
        instructor.backendId ||
        account?.backendId;

    if (!backendId) {

        alert(
            "This instructor is not linked to a Laravel account. Refresh the page and try again."
        );

        return;
    }

     * Prevent duplicate clicks while

        <span>
            Deleting...
        </span>
        `;

         * Delete the actual instructor

        const payload =
            await instructorApiRequest(

                {
                    method:
                        "DELETE"
                }
            );

         * after Laravel confirms success.

        await syncInstructorsFromApi();

        const classes =

===== LARAVEL LEARNER IMPORT IMPLEMENTATION =====
     * Required columns: Student Name, School ID.
     * Optional columns: Email, Program, Contact.
     */
    public function importLearners(Request $request): JsonResponse
    {
        if ($response = $this->checkAdmin($request)) {
            return $response;
        }

        $request->validate([
            'file' => ['required', 'file', 'mimes:xlsx', 'max:10240'],
        ]);

        try {
            $spreadsheet = IOFactory::load(
                $request->file('file')->getRealPath()
            );

            $rows = $spreadsheet->getActiveSheet()
                ->toArray(null, true, true, true);

            $spreadsheet->disconnectWorksheets();
            unset($spreadsheet);

            if (count($rows) < 2) {
                return response()->json([
                    'success' => false,
                    'message' => 'The spreadsheet contains no learner records.',
                ], 422);
            }

            $headers = array_shift($rows);
            $columns = [];

            foreach ($headers as $column => $header) {
                $key = strtolower(trim((string) $header));
                $key = preg_replace('/\s+/', ' ', $key);
                $columns[$key] = $column;
            }

            if (
                !isset($columns['student name']) ||
                !isset($columns['school id'])
            ) {
                return response()->json([
                    'success' => false,
                    'message' => 'Required columns: Student Name and School ID.',
                ], 422);
            }

            $prepared = [];
            $errors = [];
            $seenIds = [];
            $seenEmails = [];

            foreach ($rows as $index => $row) {
                $excelRow = $index + 2;
                $name = trim((string) ($row[$columns['student name']] ?? ''));
                $schoolId = trim((string) ($row[$columns['school id']] ?? ''));

                if ($name === '' && $schoolId === '') {
                    continue;
                }

                if ($name === '' || $schoolId === '') {
                    $errors[] = [
                        'row' => $excelRow,
                        'message' => 'Student Name and School ID are both required.',
                    ];
                    continue;
                }

                if (!preg_match('/^\d{2}-\d{5}$/', $schoolId)) {
                    $errors[] = [
                        'row' => $excelRow,
                        'school_id' => $schoolId,
                        'message' => 'School ID must use the format 00-00000 (two digits, a hyphen, and five digits).',
                    ];
                    continue;
                }

                $idKey = strtolower($schoolId);

                if (isset($seenIds[$idKey])) {
                    $errors[] = [
                        'row' => $excelRow,
                        'school_id' => $schoolId,
                        'message' => 'Duplicate School ID in the spreadsheet.',
                    ];
                    continue;
                }

                $seenIds[$idKey] = true;

                $getOptional = function (string $header) use ($columns, $row) {
                    if (!isset($columns[$header])) {
                        return null;
                    }

                    $value = trim((string) ($row[$columns[$header]] ?? ''));

                    return $value === '' ? null : $value;
                };

                $email = $getOptional('email');

                if ($email !== null && !filter_var($email, FILTER_VALIDATE_EMAIL)) {
                    $errors[] = [
                        'row' => $excelRow,
                        'message' => 'Invalid email address.',
                    ];
                    continue;
                }

                if ($email !== null) {
                    $emailKey = strtolower($email);

                    if (isset($seenEmails[$emailKey])) {
                        $errors[] = [
                            'row' => $excelRow,
                            'message' => 'Duplicate email address in the spreadsheet.',
                        ];
                        continue;
                    }

                    $seenEmails[$emailKey] = true;
                }

                if ($email !== null && User::where('email', $email)->exists()) {
                    $errors[] = [
                        'row' => $excelRow,
                        'message' => 'Email address already exists.',
                    ];
                    continue;
                }

                $prepared[] = [
                    'row' => $excelRow,
                    'name' => $name,
                    'school_id' => $schoolId,
                    'email' => $email,
                    'program' => $getOptional('program'),
                    'contact' => $getOptional('contact'),
                ];
            }

            $existingIds = User::whereIn(
                'school_id',
                collect($prepared)->pluck('school_id')
            )->pluck('school_id')->map(
                fn ($id) => strtolower(trim($id))
            )->all();

            foreach ($prepared as $learner) {
                if (in_array(strtolower($learner['school_id']), $existingIds, true)) {
                    $errors[] = [
                        'row' => $learner['row'],
                        'school_id' => $learner['school_id'],
                        'message' => 'School ID already exists in the system.',
                    ];
                }
            }

            if ($errors) {
                return response()->json([
                    'success' => false,
                    'message' => 'No accounts were created. Fix the spreadsheet errors and try again.',
                    'errors' => $errors,
                ], 422);
            }

            if (!$prepared) {
                return response()->json([
                    'success' => false,
                    'message' => 'No learner records were found.',
                ], 422);
            }

            $created = DB::transaction(function () use ($prepared) {
                $results = [];

                foreach ($prepared as $row) {
                    // Random, unguessable temporary credential.
                    $temporaryPassword = bin2hex(random_bytes(8));

                    $learner = User::create([
                        'name' => $row['name'],
                        'school_id' => $row['school_id'],
                        'email' => $row['email'],
                        'program' => $row['program'],
                        'contact' => $row['contact'],
                        'password' => Hash::make($temporaryPassword),
                        'role' => 'learner',
                        'password_setup_complete' => false,
                    ]);

                    $results[] = [
                        'id' => $learner->id,
                        'name' => $learner->name,
                        'school_id' => $learner->school_id,
                        'email' => $learner->email,
                        'program' => $learner->program,
                        'temporary_password' => $temporaryPassword,
                    ];
                }

                return $results;
            });

            $emailSummary = [
                'sent' => 0,
                'skipped_no_email' => 0,
                'failed' => 0,
                'not_configured' => 0,
            ];

            $mailer = (string) config('mail.default');
            $mailConfigured = false;

            if ($mailer === 'smtp') {
                $smtpHost = strtolower(trim((string) config('mail.mailers.smtp.host')));
                $smtpUsername = trim((string) config('mail.mailers.smtp.username'));
                $senderAddress = trim((string) config('mail.from.address'));

                $mailConfigured =
                    $smtpHost !== ''
                    && !in_array($smtpHost, [
                        'localhost', '127.0.0.1', '::1',
                    ], true)
                    && $senderAddress !== ''
                    && filter_var($senderAddress, FILTER_VALIDATE_EMAIL) !== false;
            } elseif (in_array($mailer, [
                'sendmail', 'mailgun', 'ses', 'ses-v2',
                'postmark', 'resend',
            ], true)) {
                $mailConfigured = true;
            }

            foreach ($created as &$learnerData) {
                if (empty($learnerData['email'])) {
                    $learnerData['email_status'] = 'skipped_no_email';
                    $emailSummary['skipped_no_email']++;
                    continue;
                }

                if (!$mailConfigured) {
                    $learnerData['email_status'] = 'not_configured';
                    $emailSummary['not_configured']++;
                    continue;
                }

                try {
                    Mail::raw(
                        "Hello {$learnerData['name']},\n\n"
                        . "Your learner account has been created.\n\n"
                        . "School ID: {$learnerData['school_id']}\n"
                        . "Temporary password: {$learnerData['temporary_password']}\n\n"
                        . "Please sign in and change your temporary password. "
                        . "Keep these credentials private.",
                        function ($message) use ($learnerData) {
